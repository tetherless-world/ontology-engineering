# Can a rigid plastic card cut a carrot, and why? (Use Case Scenario 1)
# Expected answer (from the use case): Yes.
# REQUIREMENTS under the represented target conditions.
#   Target carrot: hardness = high.
#   Required for CUT (under this target condition):

#       rigidity      = high  -> SATISFIED
#   Computed result within the TWO modeled requirements: FEASIBLE.

# ------------------------------------------------------------------------------
# Q1.1 (CQ1 - inferred judgment) - Can a rigid plastic card be used to cut a carrot?
# Retrieve the feasibility judgment assigned to the card-carrot CUT assessment.
# The inferred hasJudgment relationship indicates whether the action is feasible.
# REASONING: rule InferSatisfiedRequirementFromMappedFeatures infers that
#            both modeled CUT requirements are satisfied; rule
#            ClassifyTwoCUTRequirementsSatisfied classifies the assessment
#            as a TwoCUTRequirementsSatisfiedAssessment; rule
#            AssignFeasibleJudgment infers the hasJudgment relationship.
# RUN IN:    SNAP (Snap SPARQL Query tab, after Reasoner > HermiT > Start reasoner).
# EXPECTED (1 row, with reasoning):
#   ?judgment
#   ind:feasibleJudgment
#   Standard tab (no reasoner): NO ROWS (hasJudgment is inferred).
# ------------------------------------------------------------------------------
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>
PREFIX ind: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/individuals/>

SELECT ?judgment
WHERE {
    ind:cardCarrotCutAssessment af:hasJudgment ?judgment .
}

# ------------------------------------------------------------------------------
# Q1.2 (CQ1 - explanation) - Why does the card satisfy the modeled CUT requirements?
# Retrieve the target condition, each applicable mapping, the required feature
# level, and the card's actual level; compare them row by row.
# REASONING: asserted triples only (no reasoner needed).
# RUN IN:    STANDARD (SPARQL Query tab, no reasoner needed).
# EXPECTED (2 rows, even without a reasoner):
#   ?requirement                 ?requiredFeature  ?requiredLevel ?candidateLevel
#   af:cutEdgeThicknessRequirement  af:edgeThickness  af:low       af:low
#   af:cutRigidityRequirement       af:rigidity       af:high      af:high
#   In BOTH rows: candidate=ind:cardCandidate, target=ind:carrotTarget,
#   action=af:cut, targetFeature=af:hardness, targetLevel=af:high.
# ------------------------------------------------------------------------------
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>
PREFIX ind: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/individuals/>

SELECT ?requirement ?requiredFeature ?requiredLevel ?candidateLevel
WHERE {
  ind:cardCarrotCutAssessment af:hasCandidate ?candidate ;
                              af:hasTarget ?target ;
                              af:hasAction ?action .

  ?action af:hasRequirement ?requirement .

  ?target af:hasFeatureValue ?targetFV .
  ?targetFV af:hasFeatureType ?targetFeature ;
            af:hasLevel ?targetLevel .

  ?mapping af:usesTargetFeatureType ?targetFeature ;
           af:usesTargetLevel ?targetLevel ;
           af:affectsRequirement ?requirement ;
           af:specifiesRequiredLevel ?requiredLevel .

  ?requirement af:concernsFeature ?requiredFeature .

  OPTIONAL {
    ?candidate af:hasFeatureValue ?candidateFV .
    ?candidateFV af:hasFeatureType ?requiredFeature ;
                 af:hasLevel ?candidateLevel .
  }
}
ORDER BY ?requirement

# ==============================================================================
# CQ2: "Given the same carrot, which of a rigid plastic card, a rolling pin, a
#       kitchen sponge and a metal spatula can satisfy the CUT requirements?"
# The carrot has hardness = high, so CUT requires rigidity = high AND
# edge thickness = low.
#   card        rigidity high, edge thickness low           -> FEASIBLE
#   rolling pin rigidity high, edge thickness HIGH          -> INFEASIBLE
#   sponge      rigidity LOW,  edge thickness HIGH          -> INFEASIBLE
#   spatula     rigidity high, edge thickness NOT SUPPLIED  -> INSUFFICIENT INFO
# Answer: only the card can cut the carrot.
# ==============================================================================

# Q2.1 (MAIN CQ2 QUERY)
# Question: Which of the card, rolling pin, sponge and spatula can cut the
#           carrot? Return the inferred feasibility judgment of each object.
# RUN IN:   SNAP (HermiT started)
# EXPECTED: 4 rows
#   ind:cardCandidate        ind:feasibleJudgment
#   ind:rollingPinCandidate  ind:infeasibleJudgment
#   ind:spatulaCandidate     ind:insufficientInformationJudgment
#   ind:spongeCandidate      ind:infeasibleJudgment
# Standard tab (no reasoner): NO ROWS.
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>
PREFIX ind: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/individuals/>

SELECT ?candidate ?judgment
WHERE {
  ?assessment af:hasTarget ind:carrotTarget .
  ?assessment af:hasAction af:cut .
  ?assessment af:hasCandidate ?candidate .
  ?assessment af:hasJudgment ?judgment .
  FILTER (?candidate = ind:cardCandidate || ?candidate = ind:rollingPinCandidate ||
          ?candidate = ind:spongeCandidate || ?candidate = ind:spatulaCandidate)
}
ORDER BY ?candidate


# Q2.2
# Question: For each of the four objects, which CUT requirements are satisfied,
#           which are in conflict, and which cannot be evaluated?
# RUN IN:   SNAP (HermiT started)
# EXPECTED: 8 rows ("-" = empty cell; row order within one object may vary)
#   ?candidate               ?satisfied                       ?conflicting                     ?unresolved
#   ind:cardCandidate        af:cutEdgeThicknessRequirement   -                                -
#   ind:cardCandidate        af:cutRigidityRequirement        -                                -
#   ind:rollingPinCandidate  af:cutRigidityRequirement        -                                -
#   ind:rollingPinCandidate  -                                af:cutEdgeThicknessRequirement   -
#   ind:spatulaCandidate     af:cutRigidityRequirement        -                                -
#   ind:spatulaCandidate     -                                -                                af:cutEdgeThicknessRequirement
#   ind:spongeCandidate      -                                af:cutEdgeThicknessRequirement   -
#   ind:spongeCandidate      -                                af:cutRigidityRequirement        -
# Standard tab (no reasoner): NO ROWS.

PREFIX ind: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/individuals/>

SELECT ?candidate ?satisfied ?conflicting ?unresolved
WHERE {
  ?assessment af:hasTarget ind:carrotTarget .

# CQ3 (Priority 3) - Role D: Alex Clark
# Can a knife and a pair of scissors both meet the CUT requirements for the
# same carrot, and how does the ontology represent the difference in their
# part structures?
################################################################################
# Two equivalent combined queries are provided below:
# CQ3-A for the regular SPARQL Query view; CQ3-B for Snap SPARQL with HermiT.
# Copy ONE complete PREFIX-to-ORDER BY query at a time.
#
# PURPOSE AND LOGIC
# Compare knifeCandidate and scissorsCandidate for action cut on carrotTarget.
# Retrieve each assessment's judgment and satisfied CUT requirements, then join
# those requirements to the applicable target-feature mappings and matching
# blade feature values. Features belong to the blade parts, not the wholes.
# Optional columns show the interacting peer blade and structural classes.
# The peer must be a represented part of the same candidate. Keeping structure
# optional retains the knife even without the scissors' classifications.
#
# EXPECTED OUTPUT: SIX UNIQUE EVIDENCE ROWS
# Knife: knifeBlade x edgeThickness/rigidity requirements = two rows.
# Scissors: scissorsBlade1 and scissorsBlade2 x both requirements = four rows.
# Every row has feasibleJudgment. Required/matching levels are low for edge
# thickness and high for rigidity. Scissors rows show the other blade and the
# MultiPartObject/InteractingPartObject classes. Knife structural columns are
# unbound. Repeated judgments are expected: these are evidence rows, not six
# distinct assessments. DISTINCT removes only identical selected output rows.
#
# INTERPRETATION
# Feasibility is relative to the two modeled CUT requirements; it is not an
# empirical cutting or safety claim. The query connects assessment-level
# satisfaction to matching part evidence; it is not a complete rule-proof trace.
# One represented knife blade does not establish exactly one part under OWL's
# open-world semantics. Missing structural rows do not establish negation.
#
# ONTOLOGY SETUP AND INFERENCE PROVENANCE
# Open OE_11_TeamAffordance_Individuals.rdf and resolve its main-schema import
# to the local OE_11_TeamAffordance.rdf. Schema terms use the af: namespace and
# test individuals use the ind: namespace (both declared in each query).
# The updated shared Individuals file stores eight annotated cached conclusions:
# two judgments, four satisfaction links, and the scissors' two structural types.
# Therefore a populated result on that file does not independently prove that
# reasoning freshly derived those conclusions.
# For a fresh-inference test, use a copy of the Individuals file without the
# eight cached assertions, select HermiT, start the reasoner, and execute
# CQ3-B. Synchronize after edits.
# A Snap log identifying "Protege Null Reasoner" means HermiT is not being used.
# Without reasoning that copy returns zero combined rows; with HermiT it
# returns six.
#
################################################################################
# CQ3-A - COMBINED QUERY: REGULAR SPARQL QUERY VIEW
################################################################################
# Use the updated shared Individuals file plus its loaded main-schema import,
# or a complete local graph containing asserted and inferred axioms.
# Expected: six rows. Starting HermiT alone does not materialize conclusions in
# the regular view's asserted graph. The original unmaterialized copy gives zero
# here.

PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>
PREFIX ind: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/individuals/>

SELECT DISTINCT ?candidate ?assessment ?judgment ?requirement
                ?part ?feature ?requiredLevel ?partLevel
                ?interactingPart ?multiPartType ?interactingType
WHERE {
  ?assessment a af:Assessment ;
              af:hasCandidate ?candidate ;
              af:hasTarget ind:carrotTarget ;
              af:hasAction af:cut ;
              af:hasJudgment ?judgment ;
              af:satisfiesRequirement ?requirement .
  FILTER (?candidate = ind:knifeCandidate || ?candidate = ind:scissorsCandidate)
  af:cut af:hasRequirement ?requirement .
  ?requirement af:concernsFeature ?feature .
  ind:carrotTarget af:hasFeatureValue ?targetValue .
  ?targetValue af:hasFeatureType ?targetFeature ; af:hasLevel ?targetLevel .
  ?mapping a af:RequirementMapping ;
           af:affectsRequirement ?requirement ;
           af:usesTargetFeatureType ?targetFeature ;
           af:usesTargetLevel ?targetLevel ;
           af:specifiesRequiredLevel ?requiredLevel .
  ?candidate af:hasPart ?part .
  ?part af:hasFeatureValue ?partValue .
  ?partValue af:hasFeatureType ?feature ; af:hasLevel ?partLevel .
  FILTER (?partLevel = ?requiredLevel)
  OPTIONAL {
    ?candidate af:hasPart ?interactingPart .
    ?part af:interactsWith ?interactingPart .
    FILTER (?part != ?interactingPart)
  }
  OPTIONAL {
    ?candidate a ?multiPartType .
    FILTER (?multiPartType = af:MultiPartObject)
  }
  OPTIONAL {
    ?candidate a ?interactingType .
    FILTER (?interactingType = af:InteractingPartObject)
  }
}
ORDER BY ?candidate ?part ?requirement

PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>
PREFIX ind: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/individuals/>

SELECT DISTINCT ?candidate ?assessment ?judgment ?requirement
                ?part ?feature ?requiredLevel ?partLevel
                ?interactingPart ?multiPartType ?interactingType
WHERE {
  ?assessment a af:Assessment ;
              af:hasCandidate ?candidate ;
              af:hasTarget ind:carrotTarget ;
              af:hasAction af:cut ;
              af:hasJudgment ?judgment ;
              af:satisfiesRequirement ?requirement .
  FILTER (?candidate = ind:knifeCandidate || ?candidate = ind:scissorsCandidate)
  af:cut af:hasRequirement ?requirement .
  ?requirement af:concernsFeature ?feature .
  ind:carrotTarget af:hasFeatureValue ?targetValue .
  ?targetValue af:hasFeatureType ?targetFeature ; af:hasLevel ?targetLevel .
  ?mapping a af:RequirementMapping ;
           af:affectsRequirement ?requirement ;
           af:usesTargetFeatureType ?targetFeature ;
           af:usesTargetLevel ?targetLevel ;
           af:specifiesRequiredLevel ?requiredLevel .
  ?candidate af:hasPart ?part .
  ?part af:hasFeatureValue ?partValue .
  ?partValue af:hasFeatureType ?feature ; af:hasLevel ?partLevel .
  FILTER (?partLevel = ?requiredLevel)
  OPTIONAL {
    {
    ?candidate af:hasPart ?interactingPart .
    ?part af:interactsWith ?interactingPart .
    FILTER (?part != ?interactingPart)
    }
    UNION
    {
    ?candidate af:hasPart ?interactingPart .
    ?part af:interactsWith ?interactingPart .
    FILTER (?part != ?interactingPart)
    }
  }
  OPTIONAL {
    {
    ?candidate a ?multiPartType .
    FILTER (?multiPartType = af:MultiPartObject)
    }
    UNION
    {
    ?candidate a ?multiPartType .
    FILTER (?multiPartType = af:MultiPartObject)
    }
  }
  OPTIONAL {
    {
    ?candidate a ?interactingType .
    FILTER (?interactingType = af:InteractingPartObject)
    }
    UNION
    {
    ?candidate a ?interactingType .
    FILTER (?interactingType = af:InteractingPartObject)
    }
  }
}
ORDER BY ?candidate ?part ?requirement

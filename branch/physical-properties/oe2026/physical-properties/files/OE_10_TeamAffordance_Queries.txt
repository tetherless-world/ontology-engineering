# Can a rigid plastic card cut a carrot, and why? （Use Case Scenario 1）
# Expected answer (from the use case): Yes.
# REQUIREMENTS under the represented target conditions.
#   Target carrot: hardness = high.
#   Required for CUT (under this target condition):

#       rigidity      = high  -> SATISFIED
#   Computed result within the TWO modeled requirements: FEASIBLE.

# ------------------------------------------------------------------------------
# Q1.1 (CQ1 - inferred judgment) - Can a rigid plastic card be used to cut a carrot?
# Retrieve the feasibility judgment assigned to the card-carrot CUT assessment.
# The inferred hasJudgment relationship indicates whether the action is feasible.
# REASONING: Rule S1 infers satisfaction of the modeled CUT requirements;
#            rule S3 classifies the assessment as a
#            TwoCUTRequirementsSatisfiedAssessment;
#            rule S4 infers the hasJudgment relationship.
# RUN IN:    SNAP SPARQL Query tab, with BOTH ontologies loaded (or imports)
#            , SWRL reasoning applied, and HermiT Reasoner has been used.
# EXPECTED (1 row, with reasoning):
#   ?judgment
#   af:feasibleJudgment
#   Without reasoning, the query may return no results unless
#   the hasJudgment triple is explicitly asserted.
# ------------------------------------------------------------------------------
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>

SELECT ?judgment
WHERE {
    af:cardCarrotCutAssessment af:hasJudgment ?judgment .
}

# ------------------------------------------------------------------------------
# Q1.2 (CQ1 - explanation) - Why does the card satisfy the modeled CUT requirements?
# Retrieve the target condition, each applicable mapping, the required feature
# level, and the card's actual level; compare them row by row.
# The final OPTIONAL block also reports an ontology judgment IF it is available.
# REASONING: Asserted RDF triples for the comparisons; optional inferred
#            hasJudgment for the feasibilityStatus column.
# RUN IN:    Standard SPARQL Query tab, with BOTH ontologies loaded (or imports).
# EXPECTED (2 rows, even without a reasoner):
#   ?requirement                 ?requiredFeature  ?requiredLevel ?candidateLevel ?requirementStatus
#   af:cutEdgeThicknessRequirement  af:edgeThickness  af:low       af:low          "Satisfied"
#   af:cutRigidityRequirement       af:rigidity       af:high      af:high         "Satisfied"
#   In BOTH rows: candidate=af:cardCandidate, target=af:carrotTarget,
#   action=af:cut, targetFeature=af:hardness, targetLevel=af:high.
# ------------------------------------------------------------------------------
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>

SELECT ?candidate ?target ?action
       ?targetFeature ?targetLevel
       ?requirement ?requiredFeature ?requiredLevel
       ?candidateLevel ?requirementStatus
WHERE {
  VALUES ?assessment { af:cardCarrotCutAssessment }

  ?assessment af:hasCandidate ?candidate ;
              af:hasTarget ?target ;
              af:hasAction ?action .

  ?action af:hasRequirement ?requirement .

  ?target af:hasFeatureValue ?targetFeatureValue .
  ?targetFeatureValue af:hasFeatureType ?targetFeature ;
                      af:hasLevel ?targetLevel .

  ?mapping a af:RequirementMapping ;
           af:usesTargetFeatureType ?targetFeature ;
           af:usesTargetLevel ?targetLevel ;
           af:affectsRequirement ?requirement ;
           af:specifiesRequiredLevel ?requiredLevel .

  ?requirement af:concernsFeature ?requiredFeature .

  OPTIONAL {
    ?candidate af:hasFeatureValue ?candidateFeatureValue .
    ?candidateFeatureValue af:hasFeatureType ?requiredFeature ;
                           af:hasLevel ?candidateLevel .
  }

  BIND(
    IF(!BOUND(?candidateLevel),
       "Insufficient information",
       IF(?candidateLevel = ?requiredLevel,
          "Satisfied",
          "NotSatisfied"))
    AS ?requirementStatus
  )
}
ORDER BY ?requirement

# ------------------------------------------------------------------------------
# CQ2: "Given the same carrot, which of a rigid plastic card, a rolling pin, a
#       kitchen sponge and a metal spatula can satisfy the CUT requirements?"
# Carrot hardness = high -> CUT requires rigidity = high AND edgeThickness = low.
#   card        rigidity high, edge low          -> FEASIBLE
#   rolling pin rigidity high, edge HIGH         -> INFEASIBLE (edge conflict)
#   sponge      rigidity LOW,  edge HIGH         -> INFEASIBLE (both conflict)
#   spatula     rigidity high, edge UNSPECIFIED  -> INSUFFICIENT INFORMATION
# ------------------------------------------------------------------------------

# Q2.1 (MAIN CQ2 QUERY) - inferred judgment for each available object
# REASONING: OWL 2 DL + SWRL entailment (S1, S5, S6, S10, S4, S7, S8)
# RUN IN:    Snap SPARQL Query plugin, HermiT started
# EXPECTED (4 rows):
#   af:cardCandidate         af:feasibleJudgment
#   af:rollingPinCandidate   af:infeasibleJudgment
#   af:spatulaCandidate      af:insufficientInformationJudgment
#   af:spongeCandidate       af:infeasibleJudgment
# Without a reasoner: 0 rows.
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>

SELECT ?candidate ?judgment
WHERE {
  ?assessment rdf:type af:Assessment .
  ?assessment af:hasTarget af:carrotTarget .
  ?assessment af:hasAction af:cut .
  ?assessment af:hasCandidate ?candidate .
  ?assessment af:hasJudgment ?judgment .
  FILTER (?candidate = af:cardCandidate || ?candidate = af:rollingPinCandidate ||
          ?candidate = af:spongeCandidate || ?candidate = af:spatulaCandidate)
}
ORDER BY ?candidate


# Q2.2 - explanation: requirements each object satisfies, conflicts with, or
#        cannot be evaluated for (missing feature)
# REASONING: OWL 2 DL + SWRL entailment (S1 satisfied; S5, S6 conflict); the
#            missing feature is the asserted hasUnspecifiedFeatureType value
# RUN IN:    Snap SPARQL Query plugin, HermiT started
# EXPECTED (8 rows; "-" = unbound; row order within a candidate may vary):
#   ?candidate              ?satisfied                      ?conflicting                    ?missingFeature
#   af:cardCandidate        af:cutEdgeThicknessRequirement  -                               -
#   af:cardCandidate        af:cutRigidityRequirement       -                               -
#   af:rollingPinCandidate  -                               af:cutEdgeThicknessRequirement  -
#   af:rollingPinCandidate  af:cutRigidityRequirement       -                               -
#   af:spatulaCandidate     -                               -                               af:edgeThickness
#   af:spatulaCandidate     af:cutRigidityRequirement       -                               -
#   af:spongeCandidate      -                               af:cutEdgeThicknessRequirement  -
#   af:spongeCandidate      -                               af:cutRigidityRequirement       -
# Without a reasoner: only the spatula's missing-feature row (asserted).
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>

SELECT ?candidate ?satisfied ?conflicting ?missingFeature
WHERE {
  ?assessment af:hasTarget af:carrotTarget .
  ?assessment af:hasAction af:cut .
  ?assessment af:hasCandidate ?candidate .
  { ?assessment af:satisfiesRequirement ?satisfied . }
  UNION
  { ?assessment af:conflictsWithRequirement ?conflicting . }
  UNION
  { ?candidate af:hasUnspecifiedFeatureType ?missingFeature . }
  FILTER (?candidate = af:cardCandidate || ?candidate = af:rollingPinCandidate ||
          ?candidate = af:spongeCandidate || ?candidate = af:spatulaCandidate)
}
ORDER BY ?candidate ?satisfied ?conflicting


# Q2.3 - cross-check WITHOUT a reasoner: recompute the verdicts from the asserted
#        feature levels with the use-case evaluation rule (equal = satisfied,
#        different = conflict, no value = unresolved; any conflict -> INFEASIBLE,
#        else any unresolved -> INSUFFICIENT INFORMATION, else FEASIBLE)
# REASONING: RDF simple entailment (asserted triples), OPTIONAL / BIND / GROUP BY
# RUN IN:    standard SPARQL Query tab (no reasoner needed)
# EXPECTED (4 rows; order of the two parts inside ?explanation may vary):
#   af:cardCandidate        "edgeThickness: required low, has low (satisfied) | rigidity: required high, has high (satisfied)"      "FEASIBLE"
#   af:rollingPinCandidate  "edgeThickness: required low, has high (conflict) | rigidity: required high, has high (satisfied)"     "INFEASIBLE"
#   af:spatulaCandidate     "edgeThickness: required low, has NO VALUE (unresolved) | rigidity: required high, has high (satisfied)" "INSUFFICIENT INFORMATION"
#   af:spongeCandidate      "edgeThickness: required low, has high (conflict) | rigidity: required high, has low (conflict)"       "INFEASIBLE"
PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>

SELECT ?candidate
       (GROUP_CONCAT(?detail; separator=" | ") AS ?explanation)
       (IF(SUM(?isConflict) > 0, "INFEASIBLE",
           IF(SUM(?isUnresolved) > 0, "INSUFFICIENT INFORMATION", "FEASIBLE")) AS ?verdict)
WHERE {
  VALUES ?candidate { af:cardCandidate af:rollingPinCandidate af:spongeCandidate af:spatulaCandidate }
  ?assessment af:hasCandidate ?candidate ;
              af:hasTarget af:carrotTarget ;
              af:hasAction ?action .
  af:carrotTarget af:hasFeatureValue ?targetValue .
  ?targetValue af:hasFeatureType ?targetFeature ;
               af:hasLevel ?targetLevel .
  ?action af:hasRequirement ?requirement .
  ?requirement af:concernsFeature ?feature .
  ?mapping af:affectsRequirement ?requirement ;
           af:usesTargetFeatureType ?targetFeature ;
           af:usesTargetLevel ?targetLevel ;
           af:specifiesRequiredLevel ?requiredLevel .
  OPTIONAL {
    ?candidate af:hasFeatureValue ?candidateValue .
    ?candidateValue af:hasFeatureType ?feature ;
                    af:hasLevel ?candidateLevel .
  }
  BIND(IF(!BOUND(?candidateLevel), 1, 0) AS ?isUnresolved)
  BIND(IF(BOUND(?candidateLevel) && ?candidateLevel != ?requiredLevel, 1, 0) AS ?isConflict)
  BIND(CONCAT(STRAFTER(STR(?feature), "physical-properties/"), ": required ",
              STRAFTER(STR(?requiredLevel), "physical-properties/"), ", has ",
              IF(BOUND(?candidateLevel), STRAFTER(STR(?candidateLevel), "physical-properties/"), "NO VALUE"),
              IF(?isUnresolved = 1, " (unresolved)", IF(?isConflict = 1, " (conflict)", " (satisfied)"))) AS ?detail)
}
GROUP BY ?candidate
ORDER BY ?candidate

# Q3.2 - Explicitly distinct part pairs
# REASONING: Asserted RDF; standard SPARQL Query tab; no reasoner.

PREFIX af: <https://tw.rpi.edu/ontology-engineering/oe2026/physical-properties/>
